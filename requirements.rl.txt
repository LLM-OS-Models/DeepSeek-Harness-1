# RL extras — `uv sync`가 base harness 의존성을 푼 다음에 설치.
#
# 왜 별도 파일인가: vllm은 CUDA-specific wheel을 배포하는데, PyPI 기본
# wheel(0.25.0)은 cu130 빌드라 driver 570.86.10(CUDA 12.9)에서 동작하지
# 않는다. 그래서 GitHub release의 cu129 wheel을 직접 받아서 --no-deps로
# 설치한다. verl/deepspeed/ray는 일반 PyPI wheel.
#
# 설치 순서:
#   uv sync
#   uv pip install --no-deps -r requirements.rl.txt
#   uv pip install -r requirements.rl.runtime.txt
#
# 아래 버전은 2026-08-02에 smoke_test + GPU 6,7 롤아웃 검증을 통과한 조합.
# Python 3.12.3 / torch 2.11.0+cu128 / transformers 5.14.1 / vllm 0.25.0+cu129.

verl==0.8.0
deepspeed==0.19.3
ray==2.56.1
